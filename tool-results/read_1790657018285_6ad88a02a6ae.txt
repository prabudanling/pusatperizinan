    59→export function slugify(text: string): string {
    60→  return text
    61→    .toLowerCase()
    62→    .trim()
    63→    .replace(/\s+/g, "-")
    64→    .replace(/[^a-z0-9-]/g, "");
    65→}
    66→
    67→export function parsePrice(price: string): number {
    68→  const clean = price.replace(/\./g, "").toLowerCase();
    69→  const match = clean.match(/(\d+)(?:,(\d+))?\s*(jt|rb|juta|ribu)?/);
    70→  if (!match) return 0;
    71→  const num = parseInt(match[1] || "0", 10);
    72→  const frac = match[2] ? parseInt(match[2], 10) : 0;
    73→  let val = num + (frac ? frac / Math.pow(10, match[2]?.length || 1) : 0);
    74→  const unit = match[3];
    75→  if (unit === "jt" || unit === "juta") val *= 1_000_000;
    76→  else if (unit === "rb" || unit === "ribu") val *= 1_000;
    77→  return Math.round(val);
    78→}

   551→function buildAllPages(): ServicePage[] {
   552→  const pages: ServicePage[] = [];
   553→
   554→  // 1. Halaman induk (61 layanan)
   555→  for (const id of BASE_SERVICES.keys()) {
   556→    const p = buildBasePage(id);
   557→    if (p) pages.push(p);
   558→  }
   559→
   560→  // 2. Perizinan × 38 provinsi (layanan populer)
   561→  for (const svc of REGION_FULL_SERVICES) {
   562→    for (let i = 0; i < PROVINCES.length; i++) {
   563→      const p = buildRegionPage(svc, i);
   564→      if (p) pages.push(p);
   565→    }
   566→  }
   567→  // 2b. Perizinan lain × 8 provinsi utama
   568→  const liteServices = [...BASE_SERVICES.values()]
   569→    .filter((b) => b.category === "perizinan" && !REGION_FULL_SERVICES.includes(b.id))
   570→    .map((b) => b.id);
   571→  for (const svc of liteServices) {
   572→    for (const provName of REGION_LITE_PROVINCES) {
   573→      const idx = PROVINCES.findIndex((p) => p.name === provName);
   574→      if (idx >= 0) {
   575→        const p = buildRegionPage(svc, idx);
   576→        if (p) pages.push(p);
   577→      }
   578→    }
   579→  }
   580→
   581→  // 3. Perizinan × 10 kota besar
   582→  for (const svc of CITY_SERVICES) {
   583→    for (const city of BIG_CITIES) {
   584→      const p = buildCityPage(svc, city, "perizinan");
   585→      if (p) pages.push(p);
   586→    }
   587→  }
   588→
   589→  // 4. Pajak × 15 kota
   590→  for (const svc of TAX_CITY_SERVICES) {
   591→    for (const city of TAX_CITIES) {
   592→      const p = buildCityPage(svc, city, "pajak");
   593→      if (p) pages.push(p);
   594→    }
   595→  }
   596→
   597→  // 5. Negara PMI (17) + Negara × Sektor
   598→  for (const c of PMI_COUNTRIES) {
   599→    const cp = buildCountryPage(c.code);
   600→    if (cp) pages.push(cp);
   601→    const d = PMI_COUNTRY_MAP.get(c.code);
   602→    if (d) {
   603→      for (let i = 0; i < d.sectors.length; i++) {
   604→        const sp = buildSectorPage(c.code, i);
   605→        if (sp) pages.push(sp);
   606→      }
   607→    }
   608→  }
   609→
   610→  return pages;
   611→}

   625→/** Related: isi internal linking (induk, saudara wilayah, cross-category) */
   626→function buildRelated(): void {
   627→  const byCategory = new Map<CatalogCategory, ServicePage[]>();
   628→  const byRegion = new Map<string, ServicePage[]>();
   629→  for (const p of ALL_SERVICE_PAGES) {
   630→    const cat = byCategory.get(p.category) ?? [];
   631→    cat.push(p);
   632→    byCategory.set(p.category, cat);
   633→    if (p.region) {
   634→      const r = byRegion.get(p.region) ?? [];
   635→      r.push(p);
   636→      byRegion.set(p.region, r);
   637→    }
   638→  }
   639→
   640→  for (const p of ALL_SERVICE_PAGES) {
   641→    const rel: string[] = [];
   642→    // 1. Halaman induk (untuk kombinasi) / kombinasi populer (untuk induk)
   643→    if (p.parent) rel.push(p.parent);
   644→    const siblings = ALL_SERVICE_PAGES.filter(
   645→      (x) => x.parent === p.parent && x.kind === p.kind && x.slug !== p.slug
   646→    );
   647→    for (const s of siblings.slice(0, 2)) rel.push(s.slug);
   648→    // 2. Cross-category dari layanan dasar
   649→    const others = ALL_SERVICE_PAGES.filter(
   650→      (x) => x.kind === "base" && x.category === p.category && x.slug !== p.slug && x.slug !== p.parent
   651→    );
   652→    for (const o of others.slice(0, p.kind === "base" ? 3 : 2)) rel.push(o.slug);
   653→    // 3. Sama wilayah (untuk halaman region/city)
   654→    if (p.region) {
   655→      const sameRegion = (byRegion.get(p.region) ?? []).filter((x) => x.slug !== p.slug);
   656→      for (const s of sameRegion.slice(0, 2)) rel.push(s.slug);
   657→    }
   658→    p.related = [...new Set(rel)].slice(0, 6);
   659→  }
   660→}
